cmake_minimum_required(VERSION 3.15)
project(PureGazerDSP VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(puregazer_dsp STATIC src/GazerDSP.cpp)
set_target_properties(puregazer_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(puregazer_dsp PUBLIC src)
add_library(PureGazer SHARED src/VST2Entry.cpp)
target_link_libraries(PureGazer PRIVATE puregazer_dsp)
set_target_properties(PureGazer PROPERTIES PREFIX "" OUTPUT_NAME "PureGazer")
add_executable(puregazer_smoke tests/smoke.cpp)
target_link_libraries(puregazer_smoke PRIVATE puregazer_dsp)
enable_testing()
add_test(NAME smoke COMMAND puregazer_smoke)

add_executable(puregazer_celestial_glow tests/celestial_glow.cpp)
target_link_libraries(puregazer_celestial_glow PRIVATE puregazer_dsp)
add_test(NAME celestial_glow COMMAND puregazer_celestial_glow)
